import { expect, test } from 'bun:test';
import { Task } from '__WORKSPACE__/app/domain/task';
test('cria tarefa normalizada e registra evento', () => {
  const task = Task.create('oracle-17', '  Revisar  ');
  expect({ id: task.id, title: task.title, status: task.status }).toEqual({ id: 'oracle-17', title: 'Revisar', status: 'pending' });
  const events = task.pullEvents();
  expect(events.map(e => [e.name, e.payload])).toEqual([['TaskCreated', { taskId: 'oracle-17', title: 'Revisar' }]]);
  expect(task.pullEvents()).toEqual([]);
});
test('rejeita título vazio ou só espaços', () => {
  for (const title of ['', '   ']) expect(() => Task.create('oracle-17', title)).toThrow('Título obrigatório.');
  try { Task.create('oracle-17', ''); } catch (e) { expect((e as {code: string}).code).toBe('TASK_TITLE_REQUIRED'); }
});
