import { expect, test } from 'bun:test';
import { Task } from '__WORKSPACE__/app/domain/task';
for (const [id, title] of [['oracle-a-29', 'Auditar café'], ['oracle-b-83', 'Separar caixas']] as const) {
test('conclui, emite e rejeita repetição', () => {
  const task = Task.create(id, title); task.pullEvents(); task.complete();
  expect(task.status).toBe('completed');
  expect(task.pullEvents().map(e => [e.name, e.payload])).toEqual([['TaskCompleted', { taskId: id }]]);
  expect(() => task.complete()).toThrow();
  try { task.complete(); } catch (e) { expect((e as {code: string}).code).toBe('TASK_ALREADY_COMPLETED'); }
  expect(task.pullEvents()).toEqual([]);
});
}
