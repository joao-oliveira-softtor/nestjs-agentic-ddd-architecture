import { execFileSync } from 'node:child_process';
import { join } from 'node:path';
import { cp, mkdir } from 'node:fs/promises';
import { sha256, fingerprintOrigin } from '/home/jvtuta/code/nestjs-agentic-ddd-architecture/scripts/evals/isolation';
import { manifestSchema } from '/home/jvtuta/code/nestjs-agentic-ddd-architecture/scripts/evals/contracts';
import { campaignSchema } from '/home/jvtuta/code/nestjs-agentic-ddd-architecture/scripts/evals/aggregate';
import { loadDataset } from '/home/jvtuta/code/nestjs-agentic-ddd-architecture/scripts/evals/dataset';
import { stableStringify } from '/home/jvtuta/code/nestjs-agentic-ddd-architecture/src/compiler';
const out = process.argv[2];
if (!out) throw Error('External fresh parent required');
if (execFileSync('git',['status','--porcelain'],{encoding:'utf8'}).trim()) throw Error('Dirty origin');
const index=campaignSchema.parse(await Bun.file('evals/campaigns/cycle-20/index.json').json());
const reference=await Bun.file('evals/run.reference.json').json();
const sourceCommits=new Set<string>();
const manifestEvidence=[];
for (const slot of index.runs) {
  const path=join('evals/campaigns/cycle-20',slot.manifest.path);
  const raw=await Bun.file(path).bytes();
  if (sha256(raw)!==slot.manifest.sha256) throw Error('Raw manifest hash differs');
  const manifest=manifestSchema.parse(JSON.parse(new TextDecoder().decode(raw)));
  sourceCommits.add(manifest.sourceRef);
  if (!/^[a-f0-9]{40}$/.test(manifest.sourceRef)) throw Error('Full SHA required');
  execFileSync('git',['cat-file','-e',`${manifest.sourceRef}^{commit}`]);
  const config=reference.configurations.find((c: {id:string})=>c.id===slot.configuration);
  if (stableStringify(manifest.configurations)!==stableStringify([config])) throw Error('Configuration changed');
  if (stableStringify(manifest.budget)!==stableStringify({...reference.budget,maxInvocations:35,totalTimeoutMs:2700000})) throw Error('Budget changed');
  let count=0;
  for (const dataset of manifest.skillDatasets) {
    if (stableStringify(dataset.contextRoots)!==stableStringify(reference.skillDatasets[0].contextRoots)) throw Error('Context roots changed');
    const frozen=execFileSync('git',['show',`${manifest.sourceRef}:${dataset.path}`]);
    if (sha256(frozen)!==sha256(await Bun.file(dataset.path).bytes())) throw Error('Dataset differs from frozen source');
    count+=(await loadDataset(dataset.path)).length;
  }
  if (count!==25 || slot.plannedCases!==25) throw Error('Expected 25 questions');
  const prompt=execFileSync('git',['show',`${manifest.sourceRef}:scripts/evals/implementation.ts`],{encoding:'utf8'});
  if (!prompt.includes('{"status":"done"')) throw Error('Corrected JSON protocol absent');
  manifestEvidence.push({id:slot.id,sha256:slot.manifest.sha256,configuration:manifest.configurations[0],budget:manifest.budget});
}
if (sourceCommits.size!==1 || index.runs.length!==6) throw Error('Six runs/common source required');
execFileSync('bwrap',['--ro-bind','/','/','--unshare-pid','--unshare-ipc','--die-with-parent','--','/bin/true']);
const versions={codex:execFileSync('codex',['--version'],{encoding:'utf8'}).trim(),cursor:execFileSync('cursor-agent',['--version'],{encoding:'utf8'}).trim()};
await mkdir(join(out,'manifests'));
await cp('evals/campaigns/cycle-20/manifests',join(out,'manifests'),{recursive:true});
await Bun.write(join(out,'planned-index-before-authorization.json'),JSON.stringify(index,null,2)+'\n');
const authorized=structuredClone(index);
for (const slot of authorized.runs) slot.result={status:'not_run',reason:'authorized_not_started'};
await Bun.write(join(out,'index.json'),JSON.stringify(authorized,null,2)+'\n');
await Bun.write(join(out,'authorization-and-preflight.json'),JSON.stringify({schemaVersion:1,userReply:'autorizo',scope:'six independent issue #20 runs, three per configuration',maxNativeInvocations:210,maxTotalMinutes:270,maxInvocationsPerRun:35,maxMinutesPerRun:45,retriesOutsidePlannedSlots:false,sourceCommit:[...sourceCommits][0],runnerCommit:execFileSync('git',['rev-parse','HEAD'],{encoding:'utf8'}).trim(),originFingerprint:await fingerprintOrigin(process.cwd()),versions,manifestEvidence,positiveControl:'pending'},null,2)+'\n');
console.log('Validated six fixed manifests, 25 questions/five packets, common source, original configurations and hard budgets; isolation probe passed.');
