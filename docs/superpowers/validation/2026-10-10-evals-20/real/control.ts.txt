import { join } from 'node:path';
const {runEvaluation}=await import(`${process.cwd()}/scripts/evals/run`);
const {createScriptedAdapter}=await import(`${process.cwd()}/scripts/evals/adapters/scripted`);
const {loadDataset}=await import(`${process.cwd()}/scripts/evals/dataset`);
const {reportSchema}=await import(`${process.cwd()}/scripts/evals/report-schema`);
const {completeItem}=await import(`${process.cwd()}/test/helpers/tasks-solution`);
const {manifestSchema}=await import(`${process.cwd()}/scripts/evals/contracts`);
const root=process.argv[2];
if(!root)throw Error('New external parent required');
const manifest=manifestSchema.parse({...await Bun.file('evals/campaigns/cycle-20/manifests/codex-sol-high-r1.json').json(),configurations:[{id:'scripted-control',adapter:'scripted',model:'offline-fixture',parameters:{}}]});
const answers=[];
for(const dataset of manifest.skillDatasets) for(const c of await loadDataset(dataset.path))answers.push({finalText:JSON.stringify(c.expect)});
const report=await runEvaluation(manifest,{sourceRoot:process.cwd(),outDir:join(root,'scripted-control'),real:false,signal:new AbortController().signal,adapters:new Map([['scripted-control',createScriptedAdapter([...answers,...Array.from({length:5},()=>({finalText:'{"status":"done","summary":"offline positive control"}',before:async(req)=>{await completeItem(req.cwd,req.id)}}))])]])});
reportSchema.parse(await Bun.file(join(root,'scripted-control/report.json')).json());
if(report.status!=='completed'||report.cases.length!==25||report.cases.some(c=>c.verdict!=='correct')||report.benchmarks[0]?.items.some(i=>i.state!=='accepted')||report.benchmarks[0]?.finalVerification?.report?.status!=='done'||report.provenance.originAfter.status!=='available'||report.provenance.originAfter.value!==report.provenance.originBefore)throw Error('Positive control failed');
const preflight=await Bun.file(join(root,'authorization-and-preflight.json')).json();
if(report.provenance.originBefore!==preflight.originFingerprint||report.provenance.sourceCommit!==preflight.sourceCommit||report.provenance.runnerCommit!==preflight.runnerCommit)throw Error('Control provenance differs');
preflight.positiveControl='passed';
await Bun.write(join(root,'authorization-and-preflight.json'),JSON.stringify(preflight,null,2)+'\n');
console.log(`25 correct answers, five accepted packets and verify done; ${report.invocations} offline sessions; source fingerprint unchanged.`);
